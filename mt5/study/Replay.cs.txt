using System; using System.Collections.Generic; using System.Globalization; using System.IO; using System.Linq; using KeystoneMT5;
static class P {
  static readonly CultureInfo IC = CultureInfo.InvariantCulture;
  public static int Main(string[] a) {
    List<Mt5Bar> bars; DateTime dTo = DateTime.MinValue;
    if (Directory.Exists(a[0])) { var f0 = DateTime.ParseExact(a[1], "yyyy-MM-dd", IC); dTo = DateTime.ParseExact(a[3], "yyyy-MM-dd", IC); bars = Gold.Load(a[0], f0, dTo); foreach (var b in bars) b.Spread = 35; }
    else bars = File.ReadAllLines(a[0]).Skip(1).Select(l => l.Split(',')).Select(x => new Mt5Bar { Time = DateTime.ParseExact(x[0], "yyyy-MM-dd HH:mm:ss", IC), O = double.Parse(x[1], IC), H = double.Parse(x[2], IC), L = double.Parse(x[3], IC), C = double.Parse(x[4], IC), V = long.Parse(x[5]), Spread = 35 }).ToList();
    var day = DateTime.ParseExact(a[1], "yyyy-MM-dd", IC); var setLine = File.ReadAllLines(a[2]).Where(l => l.Contains("=")).Select(l => l.Trim()); var baseS = AutoSettings.Load(string.Join(";", setLine));
    var spec = new SimSpec { Name = "XAUUSD", Digits = 2, Point = 0.01, Contract = 100, TickValue = 1, TickSize = 0.01, VolMin = 0.01, VolStep = 0.01, VolMax = 100, NyOffsetHours = -7 };
    var variants = new List<Tuple<string, Action<AutoSettings>>> {
      Tuple.Create("YOUR SETTINGS 0.1", (Action<AutoSettings>)(s => { })),
      Tuple.Create("range+rescue@3 0.1", (Action<AutoSettings>)(s => { s.RangeOnly = true; s.BeAfter = 3; })),
      Tuple.Create("range+rescue@3 0.01/$2", (Action<AutoSettings>)(s => { s.RangeOnly = true; s.BeAfter = 3; s.EntryLots = 0.01; s.NetAdvantage = 0.02; s.Target = 2; })),
    };
    foreach (var v in variants) {
      var s = baseS.Copy(); v.Item2(s); s.TrackManual = false; s.Windows = ""; 
      var sb = new SimBroker(); var err = sb.Load(new SimSettings { Symbol = "XAUUSD", From = day, To = dTo > DateTime.MinValue ? dTo : day, Balance = a.Length > 4 ? double.Parse(a[4], IC) : 5000, CommissionPerLot = 7.84, Leverage = 500, Kind = "STANDARD", FixedSpread = 35 }, spec, bars); if (err != "") { Console.WriteLine(err); return 1; }
      var au = new HedgeAuto(); au.Send = (c, x, l) => sb.Command(c, x); sb.OnTickHook = () => au.Step(sb);
      var chk = au.Check(sb, s); if (chk != "") { Console.WriteLine(v.Item1 + ": " + chk); continue; }
      au.Start(sb, s, false); sb.Playing = true; sb.Speed = 60; double minEq = 1e9, maxGross = 0; double minMl = 1e9;
      for (int k = 0; k < 40000000 && !sb.Ended; k++) { sb.Advance(0.5); lock (sb.State.Gate) { var ac = sb.State.Account; minEq = Math.Min(minEq, ac.Equity); maxGross = Math.Max(maxGross, sb.State.Positions.Sum(p => p.Lots)); if (ac.Margin > 0) minMl = Math.Min(minMl, ac.Equity / ac.Margin * 100); } }
      var d = au.Iterations.Where(x => x.Done).ToList(); var byMonth = d.GroupBy(x => x.End.ToString("yyyy-MM")).Select(g => g.Key.Substring(5) + ":" + g.Sum(x => x.Result).ToString("0", IC)); double eq; lock (sb.State.Gate) eq = sb.State.Account.Equity;
      Console.WriteLine(string.Format(IC, "{0,-27} iter {1,3} won {2,3}  closed net {3,8:0}  equity end {4,8:0} (open {5,6:0})  lowest equity {6,7:0}  most lots {7,5:0.0}  deepest basket {8,7:0}  min margin lvl {9,6:0}%  open: {10}", v.Item1, d.Count, d.Count(x => x.Won), d.Sum(x => x.Result), eq, eq - (a.Length > 4 ? double.Parse(a[4], IC) : 5000) - d.Sum(x => x.Result), minEq, maxGross, au.Iterations.Count > 0 ? au.Iterations.Min(x => x.MaxDd) : 0, minMl, au.Current != null ? "#" + au.Current.N + " " + au.Current.AddOns + " add-ons" : "-") + "  months " + string.Join(" ", byMonth) + "  rescues " + d.Count(x => x.Reason == "RESCUE") + "  stopped: " + au.StopReason);
    }
    return 0;
  }
}
