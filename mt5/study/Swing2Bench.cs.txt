// 🪜 09o bench: the user's DEFENDER settings (his 12 Nov export) — hedge trigger LEAD vs SWING2 (N = 1 / 2 / 3) on 1-minute gold (data/MGC as XAUUSD),
// one run per period, $8,000, 💥 AUTO RESTART like the DEMO. usage: dotnet swbench.dll <LEAD|SW1|SW2|SW3[+C]> <from> <to> <balance> [candles.csv]
using System; using System.Collections.Generic; using System.Globalization; using System.IO; using System.Linq; using KeystoneMT5;
public static class P {
  static readonly CultureInfo IC = CultureInfo.InvariantCulture;
  public static int Main(string[] a) {
    string v = a[0]; var from = DateTime.ParseExact(a[1], "yyyy-MM-dd", IC); var to = DateTime.ParseExact(a[2], "yyyy-MM-dd", IC); double bal = double.Parse(a[3], IC);
    List<Mt5Bar> bars; if (a.Length > 4) bars = File.ReadAllLines(a[4]).Select(l => l.Split(',')).Where(x => x.Length >= 5 && x[0].Length >= 16 && char.IsDigit(x[0][0])).Select(x => new Mt5Bar { Time = DateTime.Parse(x[0], IC), O = double.Parse(x[1], IC), H = double.Parse(x[2], IC), L = double.Parse(x[3], IC), C = double.Parse(x[4], IC), V = x.Length > 5 ? (long)double.Parse(x[5], IC) : 0, Spread = 25 }).GroupBy(b => b.Time).Select(g => g.First()).OrderBy(b => b.Time).ToList(); else { bars = Gold.Load("/home/user/keystone--ninjatrader/data/MGC", from, to); foreach (var b in bars) b.Spread = 35; }
    var spec = new SimSpec { Name = "XAUUSD", Digits = 2, Point = 0.01, Contract = 100, TickValue = 1, TickSize = 0.01, VolMin = 0.01, VolStep = 0.01, VolMax = 100, NyOffsetHours = -7, SwapLong = -62.4, SwapShort = 28.1, SwapMode = 1, Swap3Day = DayOfWeek.Wednesday };
    var s = AutoSettings.Load(string.Join(";", File.ReadAllLines("his.txt").Where(l => l.Contains("=")).Select(l => l.Trim())));
    s.TrackManual = false; s.Windows = ""; s.OnManual = "ADOPT"; s.StepIn = false;
    if (v.StartsWith("SW")) { s.HedgeTrigger = "SWING2"; s.SwN = int.Parse(v.Substring(2, 1)); }
    if (v.Contains("+C")) s.SwBreak = "CLOSE"; if (v.Contains("+A0")) s.SwMinAtr = 0; if (v.Contains("+A1")) s.SwMinAtr = 1; if (v.Contains("+K0")) s.SwCool = 0; if (v.Contains("+K15")) s.SwCool = 15; if (v.Contains("+F")) s.SwFlip = "COUNT"; if (v.Contains("+H0")) s.SwChop = 0; if (v.Contains("+H1")) s.SwChop = 1; if (v.Contains("+H2")) s.SwChop = 2; if (v.Contains("+H3")) s.SwChop = 3;
    RadarModel.Default = RadarModel.Parse(File.ReadAllText("/home/user/keystone--ninjatrader/mt5/app/danger-model.txt"));
    var sb = new SimBroker(); var err = sb.Load(new SimSettings { Symbol = "XAUUSD", From = from, To = to, Balance = bal, CommissionPerLot = 7, Leverage = 500, Kind = "STANDARD", FixedSpread = 35, StopOutPct = 50 }, spec, bars);
    if (err != "") { Console.WriteLine("ERR " + err); return 1; }
    var au = new HedgeAuto(); au.RadarWfYear = 1; au.Send = (c, x, l) => sb.Command(c, x); sb.OnTickHook = () => au.Step(sb); au.StopOutPct = 50;
    var chk = au.Check(sb, s); if (chk != "") { Console.WriteLine(chk); return 1; }
    au.Start(sb, s, false); sb.Playing = true; sb.Speed = 60;
    int blowups = 0; double minEq = 1e9; var t0 = DateTime.Now;
    for (long k = 0; k < 80000000 && !sb.Ended; k++)
    {
      sb.Advance(0.5); double eq, b1; int open; lock (sb.State.Gate) { eq = sb.State.Account.Equity; b1 = sb.State.Account.Balance; open = sb.State.Positions.Count; } minEq = Math.Min(minEq, eq);
      bool broke = !au.Running && HedgeAuto.IsBlowReason(au.StopReason), low = open == 0 && b1 < 0.1 * bal;
      if (broke || low)
      {
        if (open > 0) { sb.Command("CLOSEALL", "XAUUSD", "ALL"); continue; }
        blowups++; lock (sb.State.Gate) sb.Account.SetBalance(bal, sb.State.ServerNow); if (au.Running) au.Stop("💥 restart"); if (au.Check(sb, s) == "") au.Start(sb, s, false);
        Console.WriteLine("BLOW " + sb.State.ServerNow.ToString("yyyy-MM-dd HH:mm", IC) + " " + b1.ToString("0", IC));
      }
      if (!au.Running && !broke) { if (au.Check(sb, s) == "") au.Start(sb, s, true); }
    }
    var its = au.Iterations.Where(x => x.Done).ToList(); double net = its.Sum(x => x.Result);
    Console.WriteLine(string.Join("\t", new object[] { "RES", v, a[1].Substring(0, 7), its.Count, its.Count(x => x.Result >= 0), net.ToString("0", IC), (minEq - bal).ToString("0", IC), blowups, its.Count(x => x.Reason == "HARD STOP"),
      its.Count > 0 ? its.Average(x => x.AddOns).ToString("0.00", IC) : "0", its.Count > 0 ? its.Average(x => (x.End - x.Start).TotalMinutes).ToString("0", IC) : "0", its.Count > 0 ? its.Min(x => x.MaxDd).ToString("0", IC) : "0", (DateTime.Now - t0).TotalSeconds.ToString("0", IC) + "s", "OPEN", au.Current != null ? au.LiveTotal().Item1.ToString("0", IC) : "0", au.Current != null ? au.Current.Start.ToString("yyyy-MM-dd", IC) : "-" }));
    if (Environment.GetEnvironmentVariable("STORY") == "1") foreach (var st in au.Story) Console.WriteLine("S " + st.Item2);
    if (Environment.GetEnvironmentVariable("TRADES") == "1") foreach (var t in au.Trades) Console.WriteLine("T " + t.Iter + " " + t.Time.ToString("MM-dd HH:mm:ss", IC) + " " + t.Side + " " + t.Lots + " @ " + t.Price + " lvl " + t.Level + " " + t.Why);
    return 0;
  }
}
