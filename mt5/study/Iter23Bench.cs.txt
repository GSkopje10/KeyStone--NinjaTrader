// with the app's own HedgeAuto on the BACKTEST SimBroker (Step + GuardStep + ProtectStep on every tick, as the app's tab does), what-if variants,
// and the same variants + scenario rules on the 6-year bench (data/MGC 1-minute gold as XAUUSD, synthetic ticks). Research only: never MT5.
// usage: dotnet it23.dll case  <variant> <export dir> [cont]            (cont = keep the automation running to the end of the data)
//        dotnet it23.dll bench <variant> <from yyyy-MM-dd> <to> <balance>
// variant = tokens joined by '+':  LEAD (as-is) | SW1 SW2 SW3 (🪜 SWING2 N) | C (SWING2 on a CLOSED candle) | STP (🚑 STEP IN sim defaults) | STP50 (MAX RISK 50 %)
//   RG (🛡 RISK GUARD LOCK) | ADV (hedge size ADVANTAGE = a fixed 0.1 lead) | SAME | D8 / D10 (max hedge distance) | HOLD (no breakeven exits: guard / rescue / BE-after off → the +$ target)
//   NOSC (no SCALE OUT at all) | BONS (no scale-out while a BREAKOUT runs) | TRAIL (breakout: no scale-out, no BE exit, trailing stop) | TRAILNC (TRAIL + no compress)
//   RE (re-entry: an iteration closed at breakeven during a breakout → the next one starts at once WITH the breakout)
//   CHOPBE (TIGHT RANGE recognised → the exit becomes breakeven at once, no wait) | XG0 (no exit guard)

{
    static readonly CultureInfo IC = CultureInfo.InvariantCulture;
    public bool BoNoScale, Trail, TrailNoCmp, ReEntry, TightAdv, TightRg; public int EodMin; public bool EodProfitOnly; public int EodCloses; public double EodSum; public double BoK = 1.0, TrailK = 2.0;
    AutoSettings baseSet; int iterN = -1, tradeIdx = 0; public double Lo = double.NaN, Hi = double.NaN; public int Hedges; bool entrySeen;
    public bool BoActive, Armed; public int BoDir; double best; DateTime lastBar = DateTime.MinValue; public double Atr5 = double.NaN, Atr1 = double.NaN;
    public int PendingReDir; public int ReEntries, Breakouts, TrailExits; public string Last = "";
    // scenario flags per iteration (recorded for every variant)
    public readonly Dictionary<int, string> Flags = new Dictionary<int, string>(); public DateTime IterStart; public double RangeAt3 = double.NaN, AtrAt3 = double.NaN; public DateTime At3;
    public double BoBest = 0;
    void Flag(int n, string f) { string s; Flags.TryGetValue(n, out s); s = s ?? ""; if (!(";" + s + ";").Contains(";" + f + ";")) Flags[n] = s == "" ? f : s + ";" + f; }

    public void Restore(HedgeAuto au)
    {
        if (baseSet == null) return; var s = au.Set;
        s.ScaleOut = baseSet.ScaleOut; s.CompressMode = baseSet.CompressMode; s.ExitGuard = baseSet.ExitGuard; s.BeAfter = baseSet.BeAfter; s.Rescue = baseSet.Rescue; s.BeLock = baseSet.BeLock; s.Target = baseSet.Target; s.DangerExit = baseSet.DangerExit; s.HedgeSize = baseSet.HedgeSize; s.RiskGuard = baseSet.RiskGuard;
    }
    public void Pre(SimBroker sb, HedgeAuto au)
    {
        if (!au.Running) return;
        if (baseSet == null) baseSet = au.Set.Copy();
        var it = au.Current;
        TrackLeft(sb);
        if (it == null || it.Done) { if (iterN != -1) { OnClosed(sb, au); Restore(au); iterN = -1; } return; }
        if (it.N != iterN)
        {
            if (iterN != -1) OnClosed(sb, au); Restore(au); iterN = it.N; Lo = Hi = double.NaN; Hedges = 0; entrySeen = false; BoActive = false; Armed = false; BoDir = 0; IterStart = it.Start; RangeAt3 = AtrAt3 = double.NaN; BoBest = 0;
            if (au.Set.FirstDir != "AUTO" && PendingReDir == 0) au.Set.FirstDir = "AUTO";
        }
        for (; tradeIdx < au.Trades.Count; tradeIdx++)
        {
            var t = au.Trades[tradeIdx]; if (t.Iter != it.N || (t.Side != "BUY" && t.Side != "SELL")) continue;
            Lo = double.IsNaN(Lo) ? t.Price : Math.Min(Lo, t.Price); Hi = double.IsNaN(Hi) ? t.Price : Math.Max(Hi, t.Price);
            if (entrySeen) Hedges++; entrySeen = true;
            if (Hedges == 3 && double.IsNaN(RangeAt3)) { RangeAt3 = Hi - Lo; AtrAt3 = Atr5; At3 = t.Time; if (!double.IsNaN(Atr5) && Atr5 > 0 && RangeAt3 < 4 * Atr5) { Flag(it.N, "TIGHT"); if (TightAdv) au.Set.HedgeSize = "ADVANTAGE"; if (TightRg) au.Set.RiskGuard = true; } }
        }
        double bid, ask; DateTime now; int open; double net;
        lock (sb.State.Gate)
        {
            var st = sb.State; now = st.ServerNow; var q = st.Quotes["XAUUSD"]; bid = q.Bid; ask = q.Ask; open = st.Positions.Count; net = st.Positions.Sum(p => p.Dir * p.Lots);
            var m1 = st.SeriesOf("XAUUSD", 1).Bars;
            if (m1.Count > 20 && m1[m1.Count - 1].Time != lastBar)
            {
                lastBar = m1[m1.Count - 1].Time;
                var m5 = st.SeriesOf("XAUUSD", 5).Bars; int e5 = m5.Count - 1; while (e5 > 0 && m5[e5].Time.AddMinutes(5) > now) e5--;
                Atr5 = TrendDetect.Atr(m5, e5, 14); int e1 = m1.Count - 1; while (e1 > 0 && m1[e1].Time.AddMinutes(1) > now) e1--; Atr1 = TrendDetect.Atr(m1, e1, 14);
            }
        }
        if (EodMin > 0 && open > 0 && au.Set.Target > -1e8)
        {   // CLOSE ALL before the daily rollover (17:00 New York = 00:00 server): EodMin minutes before it, through the engine's own close (its TAKE PROFIT path)
            var ny = now.AddHours(-7); int m = ny.Hour * 60 + ny.Minute;
            if (m >= 17 * 60 - EodMin && m < 17 * 60 && (!EodProfitOnly || au.LiveTotal().Item1 >= Math.Max(0.5, baseSet.GuardKeep))) { double totE = au.LiveTotal().Item1; EodCloses++; EodSum += totE; Flag(it.N, "EODCLOSE"); au.Set.Target = -1e9; if (Environment.GetEnvironmentVariable("VERB") == "1") Console.WriteLine("P " + now.ToString("MM-dd HH:mm:ss", IC) + " EOD CLOSE total " + totE.ToString("0.00", IC)); return; }
        }
        if (open == 0 || double.IsNaN(Atr5) || double.IsNaN(Hi)) return;
        int d = net > 0.005 ? 1 : net < -0.005 ? -1 : 0; double px = d > 0 ? bid : ask;
        // BREAKOUT from the basket's range: ≥ 2 add-ons, the net WITH the move, the price ≥ BoK × ATR(M5) beyond the range of every leg's price
        bool beyond = d != 0 && it.AddOns >= 2 && (d > 0 ? bid >= Hi + BoK * Atr5 : ask <= Lo - BoK * Atr5);
        bool inside = d == 0 || (d > 0 ? bid < Hi : ask > Lo);
        if (!BoActive && beyond) { BoActive = true; BoDir = d; best = px; Breakouts++; Flag(it.N, "BREAKOUT"); Last = now.ToString("MM-dd HH:mm", IC) + " breakout " + (d > 0 ? "UP" : "DOWN") + " range " + Lo.ToString("0.00", IC) + "–" + Hi.ToString("0.00", IC) + " ATR5 " + Atr5.ToString("0.00", IC); if (Environment.GetEnvironmentVariable("VERB") == "1") Console.WriteLine("P " + Last); Apply(au, true); }
        else if (BoActive && (inside || d != BoDir) && !Armed) { BoActive = false; Apply(au, false); if (Environment.GetEnvironmentVariable("VERB") == "1") Console.WriteLine("P " + now.ToString("MM-dd HH:mm", IC) + " breakout over (back inside)"); }
        if (!BoActive) return;
        best = BoDir > 0 ? Math.Max(best, bid) : Math.Min(best, ask); BoBest = Math.Max(BoBest, BoDir * (best - (BoDir > 0 ? Hi : Lo)) / Atr5);
        if (!Trail) return;
        double tot = au.LiveTotal().Item1;
        if (!Armed && tot >= baseSet.Target) { Armed = true; if (Environment.GetEnvironmentVariable("VERB") == "1") Console.WriteLine("P " + now.ToString("MM-dd HH:mm", IC) + " trail armed, total " + tot.ToString("0.00", IC)); }
        if (Armed && (BoDir * (px - best) <= -TrailK * Atr5 || tot <= Math.Max(0.5, baseSet.GuardKeep)))
        {   // the trailing stop: the engine's own close path (its TAKE PROFIT fires on the next step)
            au.Set.Target = -1e9; TrailExits++; Armed = false; BoActive = false;
            if (Environment.GetEnvironmentVariable("VERB") == "1") Console.WriteLine("P " + now.ToString("MM-dd HH:mm:ss", IC) + " TRAIL EXIT at " + px.ToString("0.00", IC) + " best " + best.ToString("0.00", IC) + " total " + tot.ToString("0.00", IC));
        }
    }
    // after an iteration closed while a breakout ran: how far did the price go on its way (points) in 60 / 240 minutes
    public readonly List<double[]> Left = new List<double[]>(); readonly List<double[]> leftOpen = new List<double[]>(); public readonly List<string> LeftTxt = new List<string>();
    void OnClosed(SimBroker sb, HedgeAuto au)
    {
        var last = au.Iterations.LastOrDefault(x => x.N == iterN); if (last == null || !BoActive) return;
        double bid; DateTime now; lock (sb.State.Gate) { bid = sb.State.Quotes["XAUUSD"].Bid; now = sb.State.ServerNow; }
        leftOpen.Add(new double[] { now.Ticks, bid, BoDir, 0, 0, last.Result, Atr5, iterN });
    }
    void TrackLeft(SimBroker sb)
    {
        if (leftOpen.Count == 0) return; double bid; DateTime now; lock (sb.State.Gate) { bid = sb.State.Quotes["XAUUSD"].Bid; now = sb.State.ServerNow; }
        for (int i = leftOpen.Count - 1; i >= 0; i--)
        {
            var x = leftOpen[i]; double mv = x[2] * (bid - x[1]); double mins = (now - new DateTime((long)x[0])).TotalMinutes;
            if (mins <= 60) x[3] = Math.Max(x[3], mv); x[4] = Math.Max(x[4], mv);
            if (mins >= 240) { Left.Add(x); LeftTxt.Add(new DateTime((long)x[0]).ToString("yyyy-MM-dd HH:mm", IC) + "\t" + x[5].ToString("0.00", IC) + "\t" + x[3].ToString("0.00", IC) + "\t" + x[4].ToString("0.00", IC) + "\t" + x[6].ToString("0.00", IC)); leftOpen.RemoveAt(i); }
        }
    }
    void Apply(HedgeAuto au, bool on)
    {
        var s = au.Set;
        if (BoNoScale || Trail) s.ScaleOut = on ? false : baseSet.ScaleOut;
        if (Trail) { s.ExitGuard = on ? false : baseSet.ExitGuard; s.BeAfter = on ? 0 : baseSet.BeAfter; s.Rescue = on ? false : baseSet.Rescue; s.BeLock = on ? false : baseSet.BeLock; s.Target = on ? 1e9 : baseSet.Target; s.DangerExit = on ? false : baseSet.DangerExit; }
        if (TrailNoCmp) s.CompressMode = on ? "OFF" : baseSet.CompressMode;
    }
}

    static Policy MakePolicy(string v) { var tk = v.Split('+'); var p = new Policy { TightAdv = tk.Contains("TADV"), TightRg = tk.Contains("TRG"), BoNoScale = tk.Contains("BONS"), Trail = tk.Contains("TRAIL") || tk.Contains("TRAILNC"), TrailNoCmp = tk.Contains("TRAILNC"), ReEntry = tk.Contains("RE") }; foreach (var t in tk) { if (t.StartsWith("EODP")) { p.EodMin = int.Parse(t.Substring(4)); p.EodProfitOnly = true; } else if (t.StartsWith("EOD")) p.EodMin = int.Parse(t.Substring(3)); if (t.StartsWith("K") && t.Length > 1 && char.IsDigit(t[1])) p.TrailK = double.Parse(t.Substring(1), IC); if (t.StartsWith("B") && t.Length > 1 && char.IsDigit(t[1])) p.BoK = double.Parse(t.Substring(1), IC); } return p; }

    // the guard, as the app's tab calls it (every tick) — in the bench only when it can matter (the total at / above the lowest exit level, or a new worst)
    static double lastWorstCall = 0; static int guardIter = -1;
    static void Guard(SimBroker sb, HedgeAuto au, bool everyTick)
    {
        if (everyTick) { au.GuardStep(sb, au.Set); au.ProtectStep(sb, au.Set); return; }
        if (au.Current == null) { if (!au.Running) { au.GuardStep(sb, au.Set); au.ProtectStep(sb, au.Set); } return; }
        if (au.Current.N != guardIter) { guardIter = au.Current.N; lastWorstCall = 0; au.GuardStep(sb, au.Set); return; }
        double tot = au.LiveTotal().Item1;
        if (tot >= au.Set.GuardKeep - 0.01 || tot < lastWorstCall - 5) { if (tot < lastWorstCall) lastWorstCall = tot; au.GuardStep(sb, au.Set); }
    }
